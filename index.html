const NOMBRE_CARPETA = 'Fotos_15_Eli'; // Busca automáticamente la carpeta "Fotos_15_Eli" en tu Drive (o la crea si no existe) function obtenerCarpeta() { const carpetas = DriveApp.getFoldersByName(NOMBRE_CARPETA); let folder; if (carpetas.hasNext()) { folder = carpetas.next(); } else { folder = DriveApp.createFolder(NOMBRE_CARPETA); } try { folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} return folder; } // 1. FUNCIÓN DE PRUEBA Y PERMISOS (Presiona "▷ Ejecutar" con esta función seleccionada) function autorizarPermisos() { const folder = obtenerCarpeta(); const temp = folder.createFile("prueba_conexion.txt", "Conectado correctamente"); temp.setTrashed(true); Logger.log("✅ ¡Conectado con éxito! Carpeta lista: " + folder.getUrl()); } // 2. OBTENER LAS ÚLTIMAS FOTOS DE LA CARPETA (doGet) function doGet(e) { try { const folder = obtenerCarpeta(); const files = folder.getFiles(); let photos = []; while (files.hasNext()) { let file = files.next(); let mime = file.getMimeType(); if (mime && mime.indexOf('image') !== -1) { let id = file.getId(); photos.push({ id: id, url: "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000", date: file.getDateCreated().getTime() }); } } // Ordenar de la más reciente a la más antigua photos.sort(function(a, b) { return b.date - a.date; }); const urls = photos.slice(0, 30).map(function(p) { return p.url; }); return ContentService.createTextOutput(JSON.stringify({ success: true, folderUrl: folder.getUrl(), urls: urls })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString(), urls: [] })).setMimeType(ContentService.MimeType.JSON); } } // 3. GUARDAR FOTO NUEVA EN GOOGLE DRIVE (doPost) function doPost(e) { try { const folder = obtenerCarpeta(); const rawData = e.postData.contents; const data = JSON.parse(rawData); const decoded = Utilities.base64Decode(data.base64); const blob = Utilities.newBlob(decoded, data.mimeType || 'image/jpeg', data.fileName || ('foto_' + new Date().getTime() + '.jpg')); const file = folder.createFile(blob); // Hacer que la foto sea visible para todos los invitados try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (permErr) {} const id = file.getId(); return ContentService.createTextOutput(JSON.stringify({ success: true, id: id, url: "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }

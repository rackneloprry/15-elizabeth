const NOMBRE_CARPETA = 'Fotos_15_Eli'; const NOMBRE_REPORTE = 'Reporte_CheckIn_15_Elizabeth'; function obtenerCarpetasActivas() { const iter = DriveApp.getFoldersByName(NOMBRE_CARPETA); let lista = []; while (iter.hasNext()) { let f = iter.next(); if (!f.isTrashed()) { try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} lista.push(f); } } if (lista.length === 0) { let nueva = DriveApp.createFolder(NOMBRE_CARPETA); try { nueva.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} lista.push(nueva); } return lista; } function obtenerHojaCheckIn() { const props = PropertiesService.getScriptProperties(); let sheetId = props.getProperty('CHECKIN_SHEET_ID'); let ss = null; if (sheetId) { try { ss = SpreadsheetApp.openById(sheetId); } catch (e) { ss = null; } } if (!ss) { ss = SpreadsheetApp.create(NOMBRE_REPORTE); sheetId = ss.getId(); props.setProperty('CHECKIN_SHEET_ID', sheetId); try { const file = DriveApp.getFileById(sheetId); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} } let sheet = ss.getSheets()[0]; if (sheet.getLastRow() === 0) { sheet.appendRow(['Fecha y Hora', 'ID Invitación', 'Familia / Grupo', 'Nombre del Asistente', 'Pase', 'Estado']); sheet.getRange(1, 1, 1, 6).setFontWeight('bold'); } return ss; } function autorizarPermisos() { const carpetas = obtenerCarpetasActivas(); const ss = obtenerHojaCheckIn(); Logger.log("✅ Permisos listos. Hoja de Check-In: " + ss.getUrl()); } function doGet(e) { try { const props = PropertiesService.getScriptProperties(); const galleryActive = props.getProperty('GALLERY_ACTIVE') !== 'false'; const checkIns = JSON.parse(props.getProperty('CHECKINS_DATA') || '{}'); const sheetId = props.getProperty('CHECKIN_SHEET_ID'); const sheetUrl = sheetId ? ("https://docs.google.com/spreadsheets/d/" + sheetId + "/edit") : ""; const carpetas = obtenerCarpetasActivas(); let photos = []; let vistos = {}; for (let i = 0; i < carpetas.length; i++) { let files = carpetas[i].getFiles(); while (files.hasNext()) { let file = files.next(); if (file.isTrashed()) continue; let mime = file.getMimeType(); let id = file.getId(); if (mime && mime.indexOf('image') !== -1 && !vistos[id]) { vistos[id] = true; try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} photos.push({ id: id, url: "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000", date: file.getDateCreated().getTime() }); } } } photos.sort(function(a, b) { return b.date - a.date; }); return ContentService.createTextOutput(JSON.stringify({ success: true, galleryActive: galleryActive, folderUrl: carpetas[0].getUrl(), sheetUrl: sheetUrl, checkIns: checkIns, urls: photos.slice(0, 50).map(function(p) { return p.url; }) })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString(), urls: [] })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const props = PropertiesService.getScriptProperties(); const action = data.action || 'upload'; if (action === 'checkIn') { let checkIns = JSON.parse(props.getProperty('CHECKINS_DATA') || '{}'); const key = data.key; const timestamp = data.timestamp || new Date().toLocaleString('es-SV'); if (data.checked === false) { delete checkIns[key]; } else { checkIns[key] = { grupoId: data.grupoId || '-', grupo: data.grupo || '', asistente: data.asistente || '', pase: data.pase || '1/1', hora: timestamp, estado: 'Check-In Confirmado' }; } props.setProperty('CHECKINS_DATA', JSON.stringify(checkIns)); const ss = obtenerHojaCheckIn(); const sheet = ss.getSheets()[0]; sheet.clearContents(); sheet.appendRow(['Fecha y Hora', 'ID Invitación', 'Familia / Grupo', 'Nombre del Asistente', 'Pase', 'Estado']); sheet.getRange(1, 1, 1, 6).setFontWeight('bold'); const keys = Object.keys(checkIns); for (let i = 0; i < keys.length; i++) { let item = checkIns[keys[i]]; sheet.appendRow([item.hora, item.grupoId, item.grupo, item.asistente, item.pase, item.estado]); } return ContentService.createTextOutput(JSON.stringify({ success: true, sheetUrl: ss.getUrl(), checkIns: checkIns })).setMimeType(ContentService.MimeType.JSON); } if (action === 'syncSheet') { const ss = obtenerHojaCheckIn(); const sheet = ss.getSheets()[0]; sheet.clearContents(); sheet.appendRow(['Fecha y Hora', 'ID Invitación', 'Familia / Grupo', 'Nombre del Asistente', 'Pase', 'Estado']); sheet.getRange(1, 1, 1, 6).setFontWeight('bold'); if (Array.isArray(data.rows)) { for (let i = 0; i < data.rows.length; i++) { let r = data.rows[i]; sheet.appendRow([r.hora, r.grupoId, r.grupo, r.asistente, r.pase, r.estado]); } } return ContentService.createTextOutput(JSON.stringify({ success: true, sheetUrl: ss.getUrl() })).setMimeType(ContentService.MimeType.JSON); } if (action === 'toggleGallery') { props.setProperty('GALLERY_ACTIVE', data.active ? 'true' : 'false'); return ContentService.createTextOutput(JSON.stringify({ success: true, galleryActive: data.active })).setMimeType(ContentService.MimeType.JSON); } if (action === 'delete' && data.fileId) { const file = DriveApp.getFileById(data.fileId); file.setTrashed(true); return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } const carpetas = obtenerCarpetasActivas(); const folder = carpetas[0]; const decoded = Utilities.base64Decode(data.base64); const blob = Utilities.newBlob(decoded, data.mimeType || 'image/jpeg', data.fileName || ('15Eli_' + new Date().getTime() + '.jpg')); const file = folder.createFile(blob); try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (permErr) {} const id = file.getId(); return ContentService.createTextOutput(JSON.stringify({ success: true, id: id, url: "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }

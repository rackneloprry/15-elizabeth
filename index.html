const NOMBRE_CARPETA = 'Fotos_15_Eli'; // Obtiene todas las carpetas "Fotos_15_Eli" activas (fuera de la papelera) function obtenerCarpetasActivas() { const iter = DriveApp.getFoldersByName(NOMBRE_CARPETA); let lista = []; while (iter.hasNext()) { let f = iter.next(); if (!f.isTrashed()) { try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} lista.push(f); } } if (lista.length === 0) { let nueva = DriveApp.createFolder(NOMBRE_CARPETA); try { nueva.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} lista.push(nueva); } return lista; } function autorizarPermisos() { const carpetas = obtenerCarpetasActivas(); const folder = carpetas[0]; const temp = folder.createFile("prueba.txt", "ok"); temp.setTrashed(true); Logger.log("✅ Conectado con éxito. Carpetas activas encontradas: " + carpetas.length); } // 1. LEER FOTOS DE TODAS LAS CARPETAS "Fotos_15_Eli" ACTIVAS (doGet) function doGet(e) { try { const props = PropertiesService.getScriptProperties(); const galleryActive = props.getProperty('GALLERY_ACTIVE') !== 'false'; const pinnedIds = JSON.parse(props.getProperty('PINNED_IDS') || '[]'); const carpetas = obtenerCarpetasActivas(); let photos = []; let vistos = {}; for (let i = 0; i < carpetas.length; i++) { let files = carpetas[i].getFiles(); while (files.hasNext()) { let file = files.next(); if (file.isTrashed()) continue; let mime = file.getMimeType(); let id = file.getId(); if (mime && mime.indexOf('image') !== -1 && !vistos[id]) { vistos[id] = true; try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} photos.push({ id: id, url: "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000", date: file.getDateCreated().getTime(), pinned: pinnedIds.indexOf(id) !== -1 }); } } } photos.sort(function(a, b) { return b.date - a.date; }); return ContentService.createTextOutput(JSON.stringify({ success: true, galleryActive: galleryActive, folderUrl: carpetas[0].getUrl(), photos: photos.slice(0, 50), urls: photos.slice(0, 50).map(function(p) { return p.url; }) })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString(), photos: [], urls: [] })).setMimeType(ContentService.MimeType.JSON); } } // 2. SUBIR, ELIMINAR, FIJAR O ACTIVAR/DESACTIVAR EL ÁLBUM (doPost) function doPost(e) { try { const data = JSON.parse(e.postData.contents); const props = PropertiesService.getScriptProperties(); const action = data.action || 'upload'; if (action === 'toggleGallery') { props.setProperty('GALLERY_ACTIVE', data.active ? 'true' : 'false'); return ContentService.createTextOutput(JSON.stringify({ success: true, galleryActive: data.active })).setMimeType(ContentService.MimeType.JSON); } if (action === 'delete' && data.fileId) { const file = DriveApp.getFileById(data.fileId); file.setTrashed(true); let pinnedIds = JSON.parse(props.getProperty('PINNED_IDS') || '[]'); pinnedIds = pinnedIds.filter(function(id) { return id !== data.fileId; }); props.setProperty('PINNED_IDS', JSON.stringify(pinnedIds)); return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } if (action === 'togglePin' && data.fileId) { let pinnedIds = JSON.parse(props.getProperty('PINNED_IDS') || '[]'); if (pinnedIds.indexOf(data.fileId) !== -1) { pinnedIds = pinnedIds.filter(function(id) { return id !== data.fileId; }); } else if (pinnedIds.length < 6) { pinnedIds.push(data.fileId); } props.setProperty('PINNED_IDS', JSON.stringify(pinnedIds)); return ContentService.createTextOutput(JSON.stringify({ success: true, pinnedIds: pinnedIds })).setMimeType(ContentService.MimeType.JSON); } const carpetas = obtenerCarpetasActivas(); const folder = carpetas[0]; const decoded = Utilities.base64Decode(data.base64); const blob = Utilities.newBlob(decoded, data.mimeType || 'image/jpeg', data.fileName || ('15Eli_' + new Date().getTime() + '.jpg')); const file = folder.createFile(blob); try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (permErr) {} const id = file.getId(); return ContentService.createTextOutput(JSON.stringify({ success: true, id: id, url: "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
